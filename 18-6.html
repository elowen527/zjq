图二 · 智能客服核心数据模型 ER 图（Entity Relationship） 
erDiagram
    USER ||--o{ SESSION : "发起"
    SESSION ||--|{ MESSAGE : "包含"
    MESSAGE }o--|| INTENT : "识别为"
    INTENT }o--o{ KNOWLEDGE_ITEM : "关联"
    CATEGORY ||--|{ KNOWLEDGE_ITEM : "归类"
    SESSION ||--o| TICKET : "升级为"
    AGENT ||--o{ TICKET : "处理"
    MESSAGE ||--o| FEEDBACK : "获得"

    USER {
        int user_id PK
        string nickname
        string channel "web/app/api"
        datetime register_time
    }
    SESSION {
        int session_id PK
        int user_id FK
        string status "进行中/已结束"
        datetime start_time
    }
    MESSAGE {
        int message_id PK
        int session_id FK
        string role "user/bot/agent"
        text content
        float confidence
        datetime send_time
    }
    INTENT {
        int intent_id PK
        string intent_name
        string scene
    }
    CATEGORY {
        int category_id PK
        string category_name
        int parent_id FK
    }
    KNOWLEDGE_ITEM {
        int knowledge_id PK
        int category_id FK
        string question
        text answer
        datetime update_time
    }
    TICKET {
        int ticket_id PK
        int session_id FK
        int agent_id FK
        string status "待处理/已完成"
    }
    AGENT {
        int agent_id PK
        string agent_name
        string skill_group
    }
    FEEDBACK {
        int feedback_id PK
        int message_id FK
        string type "点赞/点踩"
        text remark
    }
